use tokio::net::TcpStream;
use ratatui::{backend::CrosstermBackend, Terminal};
use optical::gpon::{OmciClient, OpticalLevels, RxPower, TxPower};

pub struct GponMonitor {
    pub ont_id: u32,
    pub rx_optical_dbm: f32,   // Target optical window: -15.0 to -24.0 dBm
    pub tx_optical_dbm: f32,
    pub laser_temp_c: f32,
    pub fec_uncorrected: u64,
    pub vlan_id: u16,          // 192.168.18.0/24 subnet mapping
}

impl GponMonitor {
    pub async fn poll_telemetry(&mut self) -> Result<OpticalLevels, NetworkError> {
        let frame = self.client.query_omci_me(0x010c).await?;
        self.rx_optical_dbm = frame.decode_rx_power();
        if self.rx_optical_dbm < -27.0 {
            println!("⚠️ High fiber attenuation on GPON loop: {:.2} dBm", self.rx_optical_dbm);
        }
        Ok(frame.levels())
    }
}
